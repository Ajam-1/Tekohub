/* ========================= GROUP CHAT ========================= */ const GROUP_MAX_MEMBERS = 50; const GROUP_MAX_TEXT = 2000; const groupSchema = new mongoose.Schema( { name: { type: String, required: true, trim: true, maxlength: 40 }, members: [{ type: mongoose.Schema.Types.ObjectId, ref: "User" }], createdBy: { type: mongoose.Schema.Types.ObjectId, ref: "User" } }, { timestamps: true } ); const groupMessageSchema = new mongoose.Schema( { groupId: { type: mongoose.Schema.Types.ObjectId, ref: "Group", required: true, index: true }, senderId: { type: mongoose.Schema.Types.ObjectId, ref: "User", required: true }, text: { type: String, required: true, trim: true, maxlength: GROUP_MAX_TEXT } }, { timestamps: true } ); const Group = mongoose.model("Group", groupSchema); const GroupMessage = mongoose.model("GroupMessage", groupMessageSchema); // Finds the group and makes sure the signed-in user is a member of it async function getMyGroup(req, res) { if (!mongoose.isValidObjectId(req.params.id)) { res.status(400).json({ message: "Invalid group." }); return null; } const group = await Group.findById(req.params.id); const myId = String(req.user.id); if (!group || !group.members.some(member => String(member) === myId)) { res.status(404).json({ message: "Group not found." }); return null; } return group; } function publicGroup(group) { return { id: group._id, name: group.name, members: group.members, createdBy: group.createdBy, isGroup: true }; } // Create a group: { name, memberIds: [...] } app.post("/groups", requireDatabase, authenticateToken, async (req, res) => { try { const myId = String(req.user.id); const name = String(req.body.name ?? "").trim(); const memberIds = Array.isArray(req.body.memberIds) ? req.body.memberIds : []; if (!name || name.length > 40) { return res.status(400).json({ message: "Give the group a name (up to 40 characters)." }); } // Remove duplicates and yourself, keep only valid ids const others = [...new Set(memberIds.map(String))] .filter(id => id !== myId && mongoose.isValidObjectId(id)); if (others.length < 2) { return res.status(400).json({ message: "A group needs at least 2 other people." }); } if (others.length + 1 > GROUP_MAX_MEMBERS) { return res.status(400).json({ message: `A group can have up to ${GROUP_MAX_MEMBERS} people.` }); } const found = await User.countDocuments({ _id: { $in: others } }); if (found !== others.length) { return res.status(400).json({ message: "Some of the selected people do not exist." }); } const group = await Group.create({ name, members: [myId, ...others], createdBy: myId }); res.status(201).json(publicGroup(group)); } catch (error) { console.error("Create group error:", error); res.status(500).json({ message: "Could not create the group." }); } }); // Groups the signed-in user belongs to app.get("/groups", requireDatabase, authenticateToken, async (req, res) => { try { const groups = await Group.find({ members: req.user.id }).sort({ updatedAt: -1 }); res.json(groups.map(publicGroup)); } catch (error) { console.error("List groups error:", error); res.status(500).json({ message: "Could not load groups." }); } }); // One group, with member names app.get("/groups/:id", requireDatabase, authenticateToken, async (req, res) => { try { const group = await getMyGroup(req, res); if (!group) { return; } const full = await Group.findById(group._id).populate("members", "username"); res.json({ id: full._id, name: full.name, createdBy: full.createdBy, isGroup: true, members: full.members.map(member => ({ id: member._id, username: member.username })) }); } catch (error) { console.error("Group details error:", error); res.status(500).json({ message: "Could not load the group." }); } }); // Latest 200 messages in a group app.get("/groups/:id/messages", requireDatabase, authenticateToken, async (req, res) => { try { const group = await getMyGroup(req, res); if (!group) { return; } // Newest 200, then put them back in oldest-to-newest order const messages = await GroupMessage.find({ groupId: group._id }) .sort({ createdAt: -1 }) .limit(200) .populate("senderId", "username"); messages.reverse(); res.json( messages.map(message => ({ id: message._id, senderId: message.senderId ? String(message.senderId._id) : "", senderName: message.senderId ? message.senderId.username : "Deleted user", text: message.text, createdAt: message.createdAt })) ); } catch (error) { console.error("Load group messages error:", error); res.status(500).json({ message: "Could not load messages." }); } }); // Send a message to a group: { text } app.post("/groups/:id/messages", requireDatabase, authenticateToken, async (req, res) => { try { const group = await getMyGroup(req, res); if (!group) { return; } const text = String(req.body.text ?? "").trim(); if (!text) { return res.status(400).json({ message: "Message cannot be empty." }); } if (text.length > GROUP_MAX_TEXT) { return res.status(400).json({ message: `Message is too long (max ${GROUP_MAX_TEXT} characters).` }); } const message = await GroupMessage.create({ groupId: group._id, senderId: req.user.id, text }); // Moves this group to the top of everyone's list await Group.findByIdAndUpdate(group._id, { $currentDate: { updatedAt: true } }); res.status(201).json({ id: message._id, senderId: String(req.user.id), text: message.text, createdAt: message.createdAt }); } catch (error) { console.error("Send group message error:", error); res.status(500).json({ message: "Could not send message." }); } }); // Add people to a group: { memberIds: [...] } (any member can add) app.post("/groups/:id/members", requireDatabase, authenticateToken, async (req, res) => { try { const group = await getMyGroup(req, res); if (!group) { return; } const memberIds = Array.isArray(req.body.memberIds) ? req.body.memberIds : []; const existing = new Set(group.members.map(String)); // Only new, valid ids const newIds = [...new Set(memberIds.map(String))] .filter(id => mongoose.isValidObjectId(id) && !existing.has(id)); if (newIds.length === 0) { return res.status(400).json({ message: "Pick at least 1 new person to add." }); } if (group.members.length + newIds.length > GROUP_MAX_MEMBERS) { return res.status(400).json({ message: `A group can have up to ${GROUP_MAX_MEMBERS} people.` }); } const found = await User.countDocuments({ _id: { $in: newIds } }); if (found !== newIds.length) { return res.status(400).json({ message: "Some of the selected people do not exist." }); } group.members.push(...newIds); await group.save(); res.json(publicGroup(group)); } catch (error) { console.error("Add members error:", error); res.status(500).json({ message: "Could not add people." }); } }); // Leave a group (a group with nobody left is deleted) app.post("/groups/:id/leave", requireDatabase, authenticateToken, async (req, res) => { try { const group = await getMyGroup(req, res); if (!group) { return; } const myId = String(req.user.id); group.members = group.members.filter(member => String(member) !== myId); if (group.members.length === 0) { await GroupMessage.deleteMany({ groupId: group._id }); await Group.deleteOne({ _id: group._id }); } else { // If the creator leaves, the oldest remaining member becomes the creator if (String(group.createdBy) === myId) { group.createdBy = group.members[0]; } await group.save(); } res.json({ message: "You left the group." }); } catch (error) { console.error("Leave group error:", error); res.status(500).json({ message: "Could not leave the group." }); } });
